WebSocket Client Code Example

const ws = new WebSocket('ws://localhost:3001');

// Connection opened
ws.addEventListener('open', (event) => {
    console.log('Connected to WebSocket server');
});

// Listen for messages
ws.addEventListener('message', (event) => {
    const data = JSON.parse(event.data);
    console.log('Message from server:', data);
    
    // Handle different message types
    if (data.type === 'TICKER_UPDATE') {
        updateTickerUI(data.symbol, data.data);
    } else if (data.type === 'OHLCV_UPDATE') {
        updateChartUI(data.symbol, data.data);
    } else if (data.type === 'RAW_DATA') {
        displayRawData(data.symbol, data.data);
    }
});

// Subscribe to a specific symbol
function subscribeToSymbol(symbol) {
    try {
        const message = JSON.stringify({
            type: 'SUBSCRIBE',
            symbol: symbol
        });
        console.log('Sending:', message);
        ws.send(message);
    } catch (error) {
        console.error('Error sending subscription:', error);
    }
}

// Unsubscribe from a symbol
function unsubscribeFromSymbol(symbol) {
    try {
        const message = JSON.stringify({
            type: 'UNSUBSCRIBE',
            symbol: symbol
        });
        console.log('Sending:', message);
        ws.send(message);
    } catch (error) {
        console.error('Error sending unsubscription:', error);
    }
}

// Connection closed
ws.addEventListener('close', (event) => {
    console.log('Disconnected from WebSocket server');
});

// Handle errors
ws.addEventListener('error', (error) => {
    console.error('WebSocket error:', error);
});
